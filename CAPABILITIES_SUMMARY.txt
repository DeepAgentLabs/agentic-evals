━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
def refund_accuracy(context):
    # Extract refund from agent execution
    refund = get_refund_from_trace(context)
    expected = 149.99 * 0.90  # 90% of order

    # Check if within $0.01 tolerance
    passed = abs(refund - expected) < 0.01

    return Score(
        name="refund_accuracy",
        value=1.0 if passed else 0.0,
        passed=passed,
        explanation=f"Refund ${refund} vs expected ${expected}"
    )

COMPLIANCE: Fraud Check Enforcement
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
def fraud_check_enforced(context):
    # Verify fraud detection was called
    compliance_called = any(
        span.tool_name == "validate_compliance"
        for span in context.sample.trace.spans
    )

    if not compliance_called:
        return Score(name="compliance", value=0, passed=False,
                    explanation="Fraud check skipped!")

    # Check result
    fraud_risk = get_fraud_risk(context)
    passed = fraud_risk == "low"

    return Score(name="compliance", value=1 if passed else 0.5,
                passed=passed, explanation=f"Risk: {fraud_risk}")

CUSTOMER SUPPORT: Tone & Professionalism
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
def professional_tone(context):
    response = context.sample.output.lower()

    # Check for professional language
    has_tone = any(word in response
                   for word in ["happy", "help", "sorry", "thanks"])

    # Penalize rude words
    has_rude = any(word in response
                   for word in ["stupid", "idiot", "refuse"])

    value = 1.0 if (has_tone and not has_rude) else 0.5

    return Score(name="tone", value=value, passed=value > 0.7,
                explanation=f"Professional tone {'detected' if has_tone else 'lacking'}")
