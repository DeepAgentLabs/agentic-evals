Deterministic Scoring

Provider-neutral rubric scoring with 10 built-in templates: factuality, translation, security, SQL, and more.

Trace-Aware

60-Second Quickstart

Your first eval in three lines

from agentic_evals import Eval, equals

Eval(
    "capitals",
    data=[
        {"input": "France", "expected": "Paris"},
        {"input": "Japan", "expected": "Tokyo"},
    ],
    task=lambda x: my_agent(x["input"]),
    scores=[equals],
)
from agentic_evals import TestSuite, TestCase, evaluate_suite

suite = TestSuite(
    name="support",
    cases=[
        TestCase(
            id="case-1",
            expected_contains=["total"],
            required_tools=["calculator"],
            max_latency_ms=1000,
        )
    ],
)

report = evaluate_suite(suite, samples)
from agentic_evals import LLMJudgeEvaluator, Score

def my_judge(context):
    correct = expected_fact in context.sample.output
    return Score(
        name="factuality",
        value=1.0 if correct else 0.0,
        passed=correct,
        explanation="Fact verified" if correct else "Hallucinated"
    )

registry.register(LLMJudgeEvaluator("fact_check", my_judge))
